import { createClient } from '@supabase/supabase-js'

// Only the public "anon" (or "publishable") key belongs here.
// Never put the service_role / secret key in this app.
const url = (import.meta.env.VITE_SUPABASE_URL as string | undefined)?.trim()
const key = (import.meta.env.VITE_SUPABASE_ANON_KEY as string | undefined)?.trim()

function findConfigProblem(): string | null {
  if (!url) return 'VITE_SUPABASE_URL is missing in the hosting settings.'
  if (!key) return 'VITE_SUPABASE_ANON_KEY is missing in the hosting settings.'
  if (!url.startsWith('https://') || url.includes('/rest/')) {
    return 'VITE_SUPABASE_URL must look like https://xxxx.supabase.co (nothing after .co).'
  }
  if (key.startsWith('sb_secret_')) {
    return 'VITE_SUPABASE_ANON_KEY holds a SECRET key. Replace it with the public anon/publishable key.'
  }
  try {
    const part = key.split('.')[1]
    if (part) {
      const payload = JSON.parse(atob(part.replace(/-/g, '+').replace(/_/g, '/')))
      if (payload.role === 'service_role') {
        return 'VITE_SUPABASE_ANON_KEY holds the service_role key. Replace it with the public anon key right away.'
      }
    }
  } catch {
    // not a readable key format; let Supabase judge it
  }
  return null
}

export const configProblem = findConfigProblem()
export const isConfigured = configProblem === null

// supabase-js adds the "apikey" header to every request automatically.
export const supabase = createClient(
  isConfigured ? (url as string) : 'http://localhost',
  isConfigured ? (key as string) : 'not-configured'
)
